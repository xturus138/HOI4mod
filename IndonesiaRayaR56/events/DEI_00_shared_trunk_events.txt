	id = dei_trunk.7

	is_triggered_only = yes

		name = dei_trunk.7.a
		add_war_support = 0.10
		add_stability = 0.05
		if = {
			limit = {
				NOT = { has_cosmetic_tag = INS_REPUBLIK_NKRI }

			}

			set_cosmetic_tag = INS_REPUBLIK_NKRI
			recruit_character = DEI_sukarno

			}

			}

				long_name = INS_PNI_party_long
			}
			set_capital = { state = 1126 }
			set_province_name = { id = 7381 name = "Jakarta" }

# --- 8. THE HAGUE ACCORD / KONFERENSI MEJA BUNDAR (KMB) ---
country_event = {
	id = dei_trunk.8
	title = dei_trunk.8.t
	desc = dei_trunk.8.d
	picture = GFX_dei_event_kmb

	trigger = {
		tag = INS
		has_war_with = HOL
		controls_state = 1126 # Batavia
		controls_state = 335  # West Java
		controls_state = 446  # East Java
		controls_state = 672  # South Sumatra
		controls_state = 334  # North Sumatra
		controls_state = 673  # West Borneo
		controls_state = 669  # Celebes
		NOT = { has_country_flag = dei_kmb_peace_signed }
	}

	mean_time_to_happen = {
		days = 7
	}

	immediate = {
		set_country_flag = dei_kmb_peace_signed
	}

	option = {
		name = dei_trunk.8.a
		if = {
			limit = { has_war_with = HOL }
			white_peace = HOL
			HOL = { white_peace = ROOT }
		}
		every_state = {
			limit = {
				is_core_of = INS
				is_owned_by = HOL
			}
			set_state_owner_to = INS
			set_state_controller_to = INS
		}
		671 = {
			add_core_of = INS
			set_state_owner_to = INS
			set_state_controller_to = INS
		}
		add_stability = 0.15
		add_war_support = 0.10
		add_political_power = 120
		news_event = { id = dei_trunk.9 hours = 6 }
	}
}
